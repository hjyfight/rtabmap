cmake_minimum_required(VERSION 3.8)
project(rtabmap_semantic)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rtabmap_conversions REQUIRED)
find_package(rtabmap_msgs REQUIRED)
find_package(rtabmap_semantic_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(RTABMap REQUIRED)
add_executable(keyframe_bridge src/keyframe_bridge.cpp)
ament_target_dependencies(keyframe_bridge rclcpp rtabmap_conversions rtabmap_msgs
  rtabmap_semantic_msgs cv_bridge sensor_msgs std_msgs RTABMap)
install(TARGETS keyframe_bridge DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/semantic_mapper DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
